cmake_minimum_required(VERSION 3.29)
project(LibertyRuntimeCoreProbe LANGUAGES C CXX OBJC OBJCXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BUILD_SHARED_LIBS OFF)
set(REXSDK "${CMAKE_CURRENT_LIST_DIR}/../upstream/glue/rexglue-sdk-main")
set(REXGLUE_ENABLE_TRACY OFF)
set(REXGLUE_ENABLE_PERF_COUNTERS ON)
set(REXGLUE_BUILD_TESTS OFF)
set(REXGLUE_USE_VULKAN ON)
set(REXGLUE_USE_D3D12 OFF)
add_compile_definitions(FMT_USE_CONSTEVAL=0 REXGLUE_BUILD_CONFIG="Release")
add_compile_options(-fno-strict-aliasing -fno-char8_t)
set(FMT_TEST OFF CACHE BOOL "" FORCE)
set(SPDLOG_FMT_EXTERNAL ON CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(SPDLOG_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${REXSDK}/thirdparty/fmt" fmt)
add_subdirectory("${REXSDK}/thirdparty/spdlog" spdlog)
set(XXHASH_BUILD_XXHSUM OFF CACHE BOOL "" FORCE)
add_subdirectory("${REXSDK}/thirdparty/xxHash/cmake_unofficial" xxhash)
add_subdirectory("${REXSDK}/thirdparty/utfcpp" utfcpp)
foreach(dep simde tomlplusplus disruptorplus)
  add_library(${dep} INTERFACE)
endforeach()
target_include_directories(simde INTERFACE "${REXSDK}/thirdparty/simde" "${REXSDK}/thirdparty/simde/simde")
target_include_directories(tomlplusplus INTERFACE "${REXSDK}/thirdparty/tomlplusplus/include")
target_include_directories(disruptorplus INTERFACE "${REXSDK}/thirdparty/disruptorplus/include")
# The upstream core CMakeLists uses PROJECT_SOURCE_DIR for its SDK-relative paths.
set(PROJECT_SOURCE_DIR "${REXSDK}")
add_subdirectory("${REXSDK}/src/core" rexcore)
# Collect the actual upstream object library, not substitute implementations.
add_library(liberty_ios_core STATIC $<TARGET_OBJECTS:rexcore>)
target_link_libraries(liberty_ios_core PRIVATE rexcore)
